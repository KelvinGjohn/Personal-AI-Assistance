{# A macro is a reusable piece of template, like a function. It draws one form field with its label and any error messages. Dropdowns pass control_class="form-select", Bootstrap's class for them. help_text adds a small hint under the field. Extra options (like autocomplete) are passed straight to the input. #} {% macro render_field(field, control_class="form-control", help_text=None) %}
{{ field(class=(control_class ~ " is-invalid") if field.errors else control_class, **kwargs) }} {% for error in field.errors %}
{{ error }}
{% endfor %} {% if help_text %}
{{ help_text }}
{% endif %}
{% endmacro %}
